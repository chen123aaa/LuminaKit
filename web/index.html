Usage

Each component is one CSS file plus an optional script. Prefer everything in a single file? glow-button/standalone.html and glass-panel/standalone.html each carry the component and a demo in one self-contained page. 

<!-- GlowButton -->
<link rel="stylesheet" href="glow-button/lumina-glow-button.css">
<script src="glow-button/lumina-glow-button.js"></script>
<button class="lg-btn">Continue</button>
<!-- GlassPanel (CSS only; the script is for the optional sheen) -->
<link rel="stylesheet" href="glass-panel/lumina-glass-panel.css">
<script src="glass-panel/lumina-glass-panel.js"></script>
<div class="lg-panel lg-panel--sheen">…</div>
/* GlassPanel tuning */
.lg-panel {
  --lgp-blur: 30;                /* backdrop blur, px */
  --lgp-saturate: 180;            /* backdrop saturation, % */
  --lgp-bg: rgba(26,28,36,.46);  /* glass tint */
  --lgp-radius: 22;               /* corner radius, px */
  --lgp-edge: 14;                 /* refracted rim, 0 disables */
  --lgp-grain: .16;               /* 0 disables grain */
}
